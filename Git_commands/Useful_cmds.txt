git clone <repository-url>
## GitHub repository ko local machine par copy (clone) karta hai.
## Ye automatically "origin" remote bhi configure kar deta hai.

git status
## Working directory aur staging area ka status dikhata hai.
## Kaunse files modified, staged ya untracked hain.

git checkout -b feature/rg-deletion
## Nayi local branch create karta hai aur usi branch par switch bhi kar deta hai.

git branch
## Local branches ki list dikhata hai.
## '*' wali current (selected) branch hoti hai.

git add .
## Current directory ke saare changes staging area me add karta hai.

git commit -m "commit remark"
## Staged changes ka local commit create karta hai.

git push
## Local branch ke commits remote repository me push karta hai.
## Ye tabhi kaam karta hai jab branch ka upstream pehle se configured ho.

git push --set-upstream origin feature/rg-deletion
# OR
git push -u origin feature/rg-deletion
## Remote repository (origin) par feature/rg-deletion branch create karta hai
## (agar pehle se nahi hai) aur local branch ko us remote branch ke saath track (upstream) kar deta hai.

git remote -v
## Configured remote repositories aur unke URLs dikhata hai.

## Example:
origin  https://github.com/creatorshubhamdevops/Terraform_Landing_Zone_Code.git (fetch)
origin  https://github.com/creatorshubhamdevops/Terraform_Landing_Zone_Code.git (push)

git fetch
## Remote repository se latest commits download karta hai.
## Local branch ko modify nahi karta.

git pull
## = git fetch + git merge
## Remote se latest changes laakar current local branch me merge karta hai.

git switch main
## Current branch se main branch par switch karta hai.

git switch feature/rg-deletion
## feature/rg-deletion branch par switch karta hai.

git branch -vv
## Local branches ke saath unka upstream (tracking branch) bhi dikhata hai.

git log --oneline --graph --all
## Branches aur commit history ko short graph format me dikhata hai.
